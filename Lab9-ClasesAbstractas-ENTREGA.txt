==============================
1. DESCRIPCION DEL PROBLEMA
==============================

Nuestro negocio es la venta de pollos, y hoy el control de clientes, empleados, ventas y gastos se lleva de forma desordenada, por lo que es difícil saber cuánto se gana realmente. Para resolverlo proponemos un sistema que guarde todo en un solo lugar y calcule la ganancia automáticamente.

Creamos la clase abstracta `Persona`, que guarda el código, el nombre y el teléfono. De ella salen `Cliente` y `Empleado`, porque los dos son personas y comparten esos tres datos, así que no hace falta repetirlos. El `Cliente` guarda además su zona de residencia y la deuda que tiene, y el `Empleado` sus días trabajados, el pago por día, los bonos y su cargo. `Persona` los obliga a implementar el método `obtenerResumen()`, que cada uno resuelve a su manera: el cliente muestra su zona y su deuda, y el empleado su cargo.

La otra clase madre es `Movimiento`, que guarda el ID, la fecha y la descripción. De ella salen `Venta` y `GastoFijo`, porque los dos son movimientos de dinero del negocio y comparten esos tres datos. `Movimiento` los obliga a implementar dos métodos: `calcularTotal()`, donde la venta se calcula por kilo y el gasto usa su monto fijo, y `esIngreso()`, que responde si el dinero entra: en la venta dice que sí y en el gasto dice que no. `Venta` guarda el cliente, el peso, el precio por kilo y el monto pagado, y su total es el peso por el precio. `GastoFijo` guarda el tipo y el monto, con los tipos implementos, combustible y peajes, agua y gas, y otros.

También creamos la interfaz `IReportable`, que usan `Empleado`, `Venta` y `GastoFijo`. No son parientes entre sí, pero las tres tienen que entregar un texto y un monto para el informe, y por eso es una interfaz y no otra clase madre.

Por último, `GestorNegocio` guarda las listas de clientes, empleados, ventas y gastos. Valida que no falten datos, que no se repita el código de un cliente o empleado y que el monto pagado no pase del total. Permite registrar, mostrar y eliminar en todas las listas, además de buscar clientes y modificar ventas y gastos. El reporte final muestra los ingresos por ventas, los gastos fijos, el pago al personal y la ganancia.

public abstract class Persona {
    private String nombre;
    private String numeroTelefono;
    private String codigoPersona;

    public Persona(String nombre, String numeroTelefono, String codigoPersona) {
        setNombre(nombre);
        setNumeroTelefono(numeroTelefono);
        this.codigoPersona = codigoPersona;
    }

    public String getNombre() {
        return nombre;
    }
    public void setNombre(String nombre) {
        if (nombre != null && !nombre.equals("")) {
            this.nombre = nombre;
        } else {
            System.out.println("El nombre es obligatorio");
        }
    }
    public String getNumeroTelefono() {
        return numeroTelefono;
    }
    public void setNumeroTelefono(String numeroTelefono) {
        if (numeroTelefono != null && !numeroTelefono.equals("")) {
            this.numeroTelefono = numeroTelefono;
        } else  {
            System.out.println("El telefono es obligatorio");
        }
    }
    public String getCodigoPersona() {
        return codigoPersona;
    }

    public abstract String obtenerResumen();

    public void mostrarDatos(){
        System.out.println("Nombre: " + getNombre());
        System.out.println("Telefono: " + getNumeroTelefono());
        System.out.println("Codigo: " + getCodigoPersona());
    }
}

public class Cliente extends Persona {
    private String zonaResidencia;
    private double deudaTotal;

    public Cliente(String nombre, String numeroTelefono, String codigoPersona, String zonaResidencia, double deudaTotal) {
        super(nombre, numeroTelefono, codigoPersona);
        setZonaResidencia(zonaResidencia);
        setDeudaTotal(deudaTotal);
    }

    public String getZonaResidencia() {
        return zonaResidencia;
    }
    public void setZonaResidencia(String zonaResidencia) {
        if (zonaResidencia != null && !zonaResidencia.equals("")) {
            this.zonaResidencia = zonaResidencia;
        } else {
            System.out.println("Debe ingresar una zona residencia");
        }
    }
    public double getDeudaTotal() {
        return deudaTotal;
    }
    public void setDeudaTotal(double deudaTotal) {
        if (deudaTotal >= 0) {
            this.deudaTotal = deudaTotal;
        } else  {
            System.out.println("La deuda no debe ser negativo");
        }
    }

    @Override
    public void mostrarDatos(){
        super.mostrarDatos();
        System.out.println("Zona Residencia: " + getZonaResidencia());
        System.out.println("Deuda Total: " + getDeudaTotal());
    }
    @Override
    public String obtenerResumen(){
        return "Cliente: " + getNombre() + " | Zona: " + zonaResidencia + " | Deuda: " + deudaTotal;
    }

    public void aumentarDeuda(double monto){
        if (monto > 0) {
            deudaTotal += monto;
        } else {
            System.out.println("El monto debe ser mayor a cero");
        }
    }
    public void reducirDeuda(double monto){
        if (monto > 0) {
            deudaTotal -= monto;
            if (deudaTotal < 0){
                deudaTotal = 0;
            }
        } else {
            System.out.println("El monto debe ser mayor a cero");
        }
    }
    public boolean tieneDeuda(){
        return deudaTotal > 0;
    }
}

========== Venta.java ==========

public class Venta extends Movimiento implements IReportable {
    private Cliente cliente;
    private double pesoKg;
    private double precioKg;
    private double montoPagado;

    public Venta(int id, String fecha, String descripcion, Cliente cliente, double pesoKg, double precioKg, double montoPagado) {
        super(id, fecha, descripcion);
        this.cliente = cliente;

        if (pesoKg > 0) {
            this.pesoKg = pesoKg;
        }
        if (precioKg > 0) {
            this.precioKg = precioKg;
        }

        double total = calcularTotal();
        if (montoPagado >= 0 && montoPagado <= total) {
            this.montoPagado = montoPagado;
        } else {
            this.montoPagado = 0;
        }

        double deudaInicial = total - this.montoPagado;
        if (this.cliente != null && deudaInicial > 0) {
            this.cliente.aumentarDeuda(deudaInicial);
        }
    }

    public Cliente getCliente() {
        return cliente;
    }

    public void setCliente(Cliente cliente) {
        this.cliente = cliente;
    }

    public double getPesoKg() {
        return pesoKg;
    }

    public void setPesoKg(double nuevoPeso) {
        if (nuevoPeso > 0) {
            double deudaAnterior = calcularDeuda();
            this.pesoKg = nuevoPeso;
            corregirDeudaCliente(deudaAnterior);
        }
    }

    public double getPrecioKg() {
        return precioKg;
    }

    public void setPrecioKg(double nuevoPrecio) {
        if (nuevoPrecio > 0) {
            double deudaAnterior = calcularDeuda();
            this.precioKg = nuevoPrecio;
            corregirDeudaCliente(deudaAnterior);
        }
    }

    public double getMontoPagado() {
        return montoPagado;
    }

    public void setMontoPagado(double montoPagado) {
        double totalActual = calcularTotal();
        if (montoPagado >= 0 && montoPagado <= totalActual) {
            double diferencia = montoPagado - this.montoPagado;
            this.montoPagado = montoPagado;
            if (cliente != null) {
                if (diferencia > 0) {
                    cliente.reducirDeuda(diferencia);
                } else if (diferencia < 0) {
                    cliente.aumentarDeuda(-diferencia);
                }
            }
        }
    }

    private void corregirDeudaCliente(double deudaAnterior) {
        double diferencia = calcularDeuda() - deudaAnterior;
        if (cliente != null) {
            if (diferencia > 0) {
                cliente.aumentarDeuda(diferencia);
            } else if (diferencia < 0) {
                cliente.reducirDeuda(-diferencia);
            }
        }
    }

    public double calcularDeuda() {
        return calcularTotal() - montoPagado;
    }

    @Override
    public double calcularTotal() {
        return pesoKg * precioKg;
    }

    @Override
    public boolean esIngreso() {
        return true;
    }

    @Override
    public String reporteTexto() {
        String nombreCliente;
        if (cliente != null) {
            nombreCliente = cliente.getNombre();
        } else {
            nombreCliente = "Sin cliente";
        }
        return "Venta realizada a " + nombreCliente + " por un total de S/ " + calcularTotal();
    }

    @Override
    public int reporteMontoMes() {
        return (int) Math.round(calcularTotal());
    }
}


========== GastoFijo.java ==========

public class GastoFijo extends Movimiento implements IReportable {
    private String tipo;
    private double monto;

    public GastoFijo(int id, String fecha, String descripcion, String tipo, double monto) {
        super(id, fecha, descripcion);
        setTipo(tipo);
        setMonto(monto);
    }

    public String getTipo() {
        return tipo;
    }

    public void setTipo(String tipo) {
        if (tipo != null && esTipoValido(tipo)) {
            this.tipo = tipo.toLowerCase().trim();
        } else {
            System.out.println("Tipo de gasto no válido. Tipos permitidos: implementos, combustible y peajes, agua y gas, otros");
        }
    }

    private boolean esTipoValido(String tipo) {
        String tipoNormalizado = tipo.toLowerCase().trim();

        if (tipoNormalizado.equals("implementos")) {
            return true;
        } else if (tipoNormalizado.equals("combustible y peajes")) {
            return true;
        } else if (tipoNormalizado.equals("agua y gas")) {
            return true;
        } else if (tipoNormalizado.equals("otros")) {
            return true;
        } else {
            return false;
        }
    }

    public double getMonto() {
        return monto;
    }

    public void setMonto(double monto) {
        if (monto >= 0) {
            this.monto = monto;
        } else {
            System.out.println("El monto no debe ser negativo");
        }
    }

    @Override
    public double calcularTotal() {
        return monto;
    }

    @Override
    public boolean esIngreso() {
        return false;
    }

    @Override
    public String reporteTexto() {
        return "Gasto Fijo (" + tipo + "): " + getDescripcion() + " - Total: S/ " + monto;
    }

    @Override
    public int reporteMontoMes() {
        return (int) Math.round(monto);
    }
}


========== GestorNegocio.java ==========

import java.util.ArrayList;

public class GestorNegocio {
    private ArrayList<Cliente> clientes = new ArrayList<>();
    private ArrayList<Empleado> empleados = new ArrayList<>();

    private ArrayList<Venta> ventas = new ArrayList<>();
    private ArrayList<GastoFijo> gastos = new ArrayList<>();
    private int siguienteIdVenta = 1;
    private int siguienteIdGasto = 1;

    public Cliente buscarCliente(String codigo) {
        for (Cliente cliente : clientes) {
            if (cliente.getCodigoPersona().equals(codigo)) {
                return cliente;
            }
        }
        return null;
    }

    public void registrarCliente(String nombre, String telefono, String codigo, String zonaResidencial, double deuda) {
        if (nombre.equals("") || telefono.equals("") || codigo.equals("") || zonaResidencial.equals("") || deuda < 0){
            System.out.println("Error: Datos incompletos o invalidos");
        } else if (buscarCliente(codigo) != null) {
            System.out.println("Error: Cliente ya existe");
        } else {
            clientes.add(new Cliente(nombre, telefono, codigo, zonaResidencial, deuda));
            System.out.println("Cliente registrado");
        }
    }

    public void mostrarClientes() {
        if (clientes.isEmpty()) {
            System.out.println("No hay clientes registrados...");
        }
        else {
            for (Cliente cliente : clientes) {
                cliente.mostrarDatos();
            }
        }
    }

    public void eliminarCliente(String codigo) {
        Cliente encontrado = buscarCliente(codigo);
        if (encontrado == null) {
            System.out.println("No se encontró el cliente");
        } else {
            for (Venta venta : ventas) {
                if (venta.getCliente() == encontrado) {
                    venta.setCliente(null);
                }
            }
            clientes.remove(encontrado);
            System.out.println("Cliente eliminado");
        }
    }

    public Empleado buscarEmpleado(String codigo) {
        for (Empleado empleado : empleados) {
            if (empleado.getCodigoPersona().equals(codigo)) {
                return empleado;
            }
        }
        return null;
    }

    public void registrarEmpleado(String nombre, String telefono, String codigo,
                                  String cargo, double pagoPorDia,
                                  int diasTrabajados, double bono) {
        if (nombre.equals("") || telefono.equals("") || codigo.equals("") || cargo.equals("")
                || pagoPorDia <= 0 || diasTrabajados < 0 || bono < 0) {
            System.out.println("Error: Datos incompletos o inválidos");
        } else if (buscarEmpleado(codigo) != null) {
            System.out.println("Error: Empleado ya existe");
        } else {
            empleados.add(new Empleado(nombre, telefono, codigo, cargo,
                    pagoPorDia, diasTrabajados, bono));
            System.out.println("Empleado registrado");
        }
    }

    public void eliminarEmpleado(String codigo) {
        Empleado encontrado = buscarEmpleado(codigo);
        if (encontrado == null) {
            System.out.println("No se encontró el empleado");
        } else {
            empleados.remove(encontrado);
            System.out.println("Empleado eliminado");
        }
    }

    public void mostrarEmpleados() {
        if (empleados.isEmpty()) {
            System.out.println("No hay empleados registrados...");
        } else {
            for (Empleado empleado : empleados) {
                System.out.println(empleado.reporteTexto());
            }
        }
    }

    public Venta buscarVenta(int id) {
        for (Venta venta : ventas) {
            if (venta.getId() == id) {
                return venta;
            }
        }
        return null;
    }

    public void registrarVenta(String fecha, String descripcion, String codigoCliente, double pesoKg, double precioKg, double montoPagado) {
        if (fecha.equals("") || descripcion.equals("") || pesoKg <= 0 || precioKg <= 0 || montoPagado < 0) {
            System.out.println("Error: Datos incompletos o inválidos");
            return;
        }

        double total = pesoKg * precioKg;
        if (montoPagado > total) {
            System.out.println("Error: El monto pagado no puede ser mayor al total de la venta");
            return;
        }

        Cliente cliente = null;
        if (!codigoCliente.equals("")) {
            cliente = buscarCliente(codigoCliente);
            if (cliente == null) {
                System.out.println("Cliente no encontrado, la venta se registrará sin cliente asociado");
            }
        }

        Venta venta = new Venta(siguienteIdVenta, fecha, descripcion, cliente, pesoKg, precioKg, montoPagado);
        ventas.add(venta);
        siguienteIdVenta = siguienteIdVenta + 1;
        System.out.println("Venta registrada");
    }

    public void mostrarVentas() {
        if (ventas.isEmpty()) {
            System.out.println("No hay ventas registradas...");
        } else {
            for (Venta venta : ventas) {
                System.out.println(venta.reporteTexto());
            }
        }
    }

    public void modificarVenta(int id, double nuevoMontoPagado) {
        Venta encontrada = buscarVenta(id);
        if (encontrada == null) {
            System.out.println("No se encontró la venta");
        } else if (nuevoMontoPagado < 0) {
            System.out.println("Error: El monto pagado no debe ser negativo");
        } else if (nuevoMontoPagado > encontrada.calcularTotal()) {
            System.out.println("Error: El monto pagado no puede ser mayor al total de la venta");
        } else {
            encontrada.setMontoPagado(nuevoMontoPagado);
            System.out.println("Venta modificada");
        }
    }

    public void eliminarVenta(int id) {
        Venta encontrada = buscarVenta(id);
        if (encontrada == null) {
            System.out.println("No se encontró la venta");
        } else {
            if (encontrada.getCliente() != null) {
                encontrada.getCliente().reducirDeuda(encontrada.calcularDeuda());
            }
            ventas.remove(encontrada);
            System.out.println("Venta eliminada");
        }
    }

    public GastoFijo buscarGasto(int id) {
        for (GastoFijo gasto : gastos) {
            if (gasto.getId() == id) {
                return gasto;
            }
        }
        return null;
    }

    public void registrarGasto(String fecha, String descripcion, String tipo, double monto) {
        if (fecha.equals("") || descripcion.equals("") || tipo.equals("") || monto <= 0) {
            System.out.println("Error: Datos incompletos o inválidos");
        } else {
            GastoFijo gasto = new GastoFijo(siguienteIdGasto, fecha, descripcion, tipo, monto);

            if (gasto.getTipo() == null) {
                System.out.println("Error: El gasto no se registró");
            } else {
                gastos.add(gasto);
                siguienteIdGasto = siguienteIdGasto + 1;
                System.out.println("Gasto fijo registrado");
            }
        }
    }

    public void mostrarGastos() {
        if (gastos.isEmpty()) {
            System.out.println("No hay gastos registrados...");
        } else {
            for (GastoFijo gasto : gastos) {
                System.out.println(gasto.reporteTexto());
            }
        }
    }

    public void modificarGasto(int id, double nuevoMonto) {
        GastoFijo encontrado = buscarGasto(id);
        if (encontrado == null) {
            System.out.println("No se encontró el gasto");
        } else if (nuevoMonto <= 0) {
            System.out.println("Error: El monto debe ser mayor a cero");
        } else {
            encontrado.setMonto(nuevoMonto);
            System.out.println("Gasto modificado");
        }
    }

    public void eliminarGasto(int id) {
        GastoFijo encontrado = buscarGasto(id);
        if (encontrado == null) {
            System.out.println("No se encontró el gasto");
        } else {
            gastos.remove(encontrado);
            System.out.println("Gasto eliminado");
        }
    }

    public void mostrarPersonasYMovimientos() {
        ArrayList<Persona> personas = new ArrayList<>();
        for (Cliente cliente : clientes) {
            personas.add(cliente);
        }
        for (Empleado empleado : empleados) {
            personas.add(empleado);
        }

        ArrayList<Movimiento> movimientos = new ArrayList<>();
        for (Venta venta : ventas) {
            movimientos.add(venta);
        }
        for (GastoFijo gasto : gastos) {
            movimientos.add(gasto);
        }

        System.out.println("--- Personas ---");
        if (personas.isEmpty()) {
            System.out.println("No hay personas registradas...");
        } else {
            for (Persona persona : personas) {
                System.out.println(persona.obtenerResumen());
            }
        }

        System.out.println("--- Movimientos ---");
        if (movimientos.isEmpty()) {
            System.out.println("No hay movimientos registrados...");
        } else {
            for (Movimiento movimiento : movimientos) {
                System.out.println(movimiento.mostrarMovimiento() + " | Total: S/ " + movimiento.calcularTotal());
            }
        }
    }

    public void mostrarReportes() {
        ArrayList<IReportable> reportables = new ArrayList<>();
        for (Empleado empleado : empleados) {
            reportables.add(empleado);
        }
        for (Venta venta : ventas) {
            reportables.add(venta);
        }
        for (GastoFijo gasto : gastos) {
            reportables.add(gasto);
        }

        if (reportables.isEmpty()) {
            System.out.println("No hay datos para el reporte...");
            return;
        }

        System.out.println("--- REPORTE DEL MES ---");
        for (IReportable reportable : reportables) {
            System.out.println(reportable.reporteTexto());
        }

        int totalVentas = 0;
        for (Venta venta : ventas) {
            totalVentas = totalVentas + venta.reporteMontoMes();
        }
        int totalGastos = 0;
        for (GastoFijo gasto : gastos) {
            totalGastos = totalGastos + gasto.reporteMontoMes();
        }
        int totalSueldos = 0;
        for (Empleado empleado : empleados) {
            totalSueldos = totalSueldos + empleado.reporteMontoMes();
        }

        System.out.println("Ingresos por ventas: S/ " + totalVentas);
        System.out.println("Gastos fijos: S/ " + totalGastos);
        System.out.println("Pago al personal: S/ " + totalSueldos);
        System.out.println("Ganancia del mes: S/ " + (totalVentas - totalGastos - totalSueldos));
    }
}
