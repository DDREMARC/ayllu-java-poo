Nuestro negocio es la venta de pollos. Antes, el control de clientes, empleados, ventas y gastos se llevaba de forma desordenada, y no era facil saber cuanto se ganaba realmente. Este sistema guarda todo en una base de datos MySQL y calcula la ganancia automaticamente.

2. OBJETIVO
Unir en un solo sistema los temas vistos en el curso:
- Laboratorio 9: clases abstractas (Persona, Movimiento), herencia, polimorfismo e interfaz (IReportable).
- Laboratorio 10: conexion JDBC a MySQL, sentencias Statement, CRUD completo y script SQL.
- Laboratorio 11: sentencias PreparedStatement y arquitectura MVC (Modelo, Vista, Controlador).

3. BASE DE DATOS (ayllu_pollos)
- personas: clientes y empleados en una sola tabla. Clave primaria codigo_persona. La columna es_empleado indica el tipo (1 empleado, 0 cliente).
- tipo_gasto: catalogo de tipos de gasto fijo.
- gasto_fijo: gastos del negocio. Clave primaria id_gasto. Clave foranea id_tipo hacia tipo_gasto.
- ventas: ventas realizadas. Clave primaria Codigo_venta. Clave foranea Codigo_Cliente hacia personas.
Diagrama entidad-relacion: proyecto-final/docs/MER.svg

4. ESTRUCTURA DEL CODIGO (arquitectura MVC)
- modelo: representa las filas de las tablas. Persona y Movimiento son clases abstractas. Cliente y Empleado heredan de Persona. Venta y GastoFijo heredan de Movimiento. IReportable es una interfaz que usan Empleado, Venta y GastoFijo para el informe.
- controlador: ejecuta el SQL con PreparedStatement y devuelve datos. No imprime nada. ReporteController calcula ganancias y planilla usando polimorfismo.
- vista: muestra los menus, lee las opciones y muestra mensajes. Lector hace las lecturas seguras por consola.
- conexion: DatabaseConnector abre y cierra la conexion a la base de datos.

5. REGLAS DE NEGOCIO
- No se puede registrar una persona con un codigo que ya existe.
- Una venta solo se registra para un cliente existente. El monto pagado no puede superar el total.
- Si el cliente paga menos del total, la diferencia se suma a su deuda. Al actualizar o eliminar la venta, la deuda se corrige.
- No se puede eliminar un cliente que tiene ventas registradas.
- Ganancia real (caja) = total cobrado - gastos fijos.
- Ganancia proyectada = total vendido - gastos fijos.
- Pago de planilla = dias trabajados x pago por dia + bono.

6. COMO EJECUTAR
1. Iniciar MySQL o MariaDB (XAMPP).
2. Ejecutar sql/script_ayllu_pollos.sql en phpMyAdmin.
3. Ejecutar ejecutar.bat. Compila el proyecto y arranca el programa. El driver de MySQL esta en lib/.

7. CONTENIDO DE ESTE DOCUMENTO
Este documento contiene el contexto, todas las clases Java del proyecto y el script SQL, en ese orden.

        }

package controlador;

import modelo.GastoFijo;
import modelo.TipoGasto;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

public class GastoFijoController {
    private Connection connection;

    private static final String SELECT_BASE =
            "SELECT g.id_gasto, g.fecha, g.monto, t.id_tipo, t.nombre "
            + "FROM gasto_fijo g INNER JOIN tipo_gasto t ON g.id_tipo = t.id_tipo ";

    public GastoFijoController(Connection connection) {
        this.connection = connection;
    }

    private GastoFijo construir(ResultSet rs) throws SQLException {
        TipoGasto tipo = new TipoGasto(rs.getInt("id_tipo"), rs.getString("nombre"));
        return new GastoFijo(rs.getInt("id_gasto"), rs.getString("fecha"), tipo, rs.getDouble("monto"));
    }

    public List<TipoGasto> listarTipos() throws SQLException {
        List<TipoGasto> tipos = new ArrayList<>();
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement("SELECT id_tipo, nombre FROM tipo_gasto ORDER BY id_tipo");
            rs = ps.executeQuery();
            while (rs.next()) {
                tipos.add(new TipoGasto(rs.getInt("id_tipo"), rs.getString("nombre")));
            }
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
        return tipos;
    }

    public TipoGasto buscarTipo(int idTipo) throws SQLException {
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement("SELECT id_tipo, nombre FROM tipo_gasto WHERE id_tipo = ?");
            ps.setInt(1, idTipo);
            rs = ps.executeQuery();
            if (rs.next()) {
                return new TipoGasto(rs.getInt("id_tipo"), rs.getString("nombre"));
            }
            return null;
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
    }

    public List<GastoFijo> listar() throws SQLException {
        List<GastoFijo> lista = new ArrayList<>();
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(SELECT_BASE + "ORDER BY g.fecha, g.id_gasto");
            rs = ps.executeQuery();
            while (rs.next()) {
                lista.add(construir(rs));
            }
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
        return lista;
    }

    public GastoFijo buscar(int idGasto) throws SQLException {
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(SELECT_BASE + "WHERE g.id_gasto = ?");
            ps.setInt(1, idGasto);
            rs = ps.executeQuery();
            if (rs.next()) {
                return construir(rs);
            }
            return null;
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
    }

    public int registrar(GastoFijo gasto) throws SQLException {
        String sql = "INSERT INTO gasto_fijo (id_tipo, fecha, monto) VALUES (?, ?, ?)";
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setInt(1, gasto.getTipo().getIdTipo());
            ps.setString(2, gasto.getFecha());
            ps.setDouble(3, gasto.getMonto());
            return ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
    }

    public int actualizar(GastoFijo gasto) throws SQLException {
        String sql = "UPDATE gasto_fijo SET id_tipo = ?, fecha = ?, monto = ? WHERE id_gasto = ?";
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setInt(1, gasto.getTipo().getIdTipo());
            ps.setString(2, gasto.getFecha());
            ps.setDouble(3, gasto.getMonto());
            ps.setInt(4, gasto.getId());
            return ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
    }

    public int eliminar(int idGasto) throws SQLException {
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement("DELETE FROM gasto_fijo WHERE id_gasto = ?");
            ps.setInt(1, idGasto);
            return ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
    }
}

package controlador;

import modelo.Cliente;
import modelo.Empleado;
import modelo.Persona;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

public class PersonaController {
    private Connection connection;

    public PersonaController(Connection connection) {
        this.connection = connection;
    }

    private Persona construir(ResultSet rs) throws SQLException {
        String codigo = rs.getString("codigo_persona");
        String nombre = rs.getString("nombre");
        String telefono = rs.getString("numero_telefono");
        if (rs.getInt("es_empleado") == 1) {
            return new Empleado(codigo, nombre, telefono,
                    rs.getString("cargo"), rs.getDouble("pago_por_dia"),
                    rs.getInt("dias_trabajados"), rs.getDouble("bono"));
        }
        return new Cliente(codigo, nombre, telefono,
                rs.getString("zona_residencia"), rs.getDouble("deuda_total"));
    }

    public List<Persona> listar() throws SQLException {
        List<Persona> lista = new ArrayList<>();
        String sql = "SELECT * FROM personas ORDER BY es_empleado, codigo_persona";
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            rs = ps.executeQuery();
            while (rs.next()) {
                lista.add(construir(rs));
            }
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
        return lista;
    }

    public Persona buscar(String codigo) throws SQLException {
        String sql = "SELECT * FROM personas WHERE codigo_persona = ?";
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setString(1, codigo);
            rs = ps.executeQuery();
            if (rs.next()) {
                return construir(rs);
            }
            return null;
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
    }

    public Cliente buscarCliente(String codigo) throws SQLException {
        Persona p = buscar(codigo);
        if (p instanceof Cliente) {
            return (Cliente) p;
        }
        return null;
    }

    public int registrar(Persona persona) throws SQLException {
        PreparedStatement ps = null;
        try {
            if (persona.esEmpleado()) {
                Empleado e = (Empleado) persona;
                String sql = "INSERT INTO personas (codigo_persona, nombre, numero_telefono, es_empleado, "
                        + "cargo, pago_por_dia, dias_trabajados, bono) VALUES (?, ?, ?, 1, ?, ?, ?, ?)";
                ps = connection.prepareStatement(sql);
                ps.setString(1, e.getCodigoPersona());
                ps.setString(2, e.getNombre());
                ps.setString(3, e.getNumeroTelefono());
                ps.setString(4, e.getCargo());
                ps.setDouble(5, e.getPagoPorDia());
                ps.setInt(6, e.getDiasTrabajados());
                ps.setDouble(7, e.getBono());
            } else {
                Cliente c = (Cliente) persona;
                String sql = "INSERT INTO personas (codigo_persona, nombre, numero_telefono, es_empleado, "
                        + "zona_residencia, deuda_total) VALUES (?, ?, ?, 0, ?, ?)";
                ps = connection.prepareStatement(sql);
                ps.setString(1, c.getCodigoPersona());
                ps.setString(2, c.getNombre());
                ps.setString(3, c.getNumeroTelefono());
                ps.setString(4, c.getZonaResidencia());
                ps.setDouble(5, c.getDeudaTotal());
            }
            return ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
    }

    public int actualizar(Persona persona) throws SQLException {
        PreparedStatement ps = null;
        try {
            if (persona.esEmpleado()) {
                Empleado e = (Empleado) persona;
                String sql = "UPDATE personas SET nombre = ?, numero_telefono = ?, cargo = ?, pago_por_dia = ?, "
                        + "dias_trabajados = ?, bono = ? WHERE codigo_persona = ? AND es_empleado = 1";
                ps = connection.prepareStatement(sql);
                ps.setString(1, e.getNombre());
                ps.setString(2, e.getNumeroTelefono());
                ps.setString(3, e.getCargo());
                ps.setDouble(4, e.getPagoPorDia());
                ps.setInt(5, e.getDiasTrabajados());
                ps.setDouble(6, e.getBono());
                ps.setString(7, e.getCodigoPersona());
            } else {
                Cliente c = (Cliente) persona;
                String sql = "UPDATE personas SET nombre = ?, numero_telefono = ?, zona_residencia = ?, "
                        + "deuda_total = ? WHERE codigo_persona = ? AND es_empleado = 0";
                ps = connection.prepareStatement(sql);
                ps.setString(1, c.getNombre());
                ps.setString(2, c.getNumeroTelefono());
                ps.setString(3, c.getZonaResidencia());
                ps.setDouble(4, c.getDeudaTotal());
                ps.setString(5, c.getCodigoPersona());
            }
            return ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
    }

    public int eliminar(String codigo) throws SQLException {
        String sql = "DELETE FROM personas WHERE codigo_persona = ?";
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setString(1, codigo);

                ps.close();
            }
        }
    }

        String sql = "SELECT COUNT(*) AS total FROM ventas WHERE Codigo_Cliente = ?";
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);

            if (rs.next()) {

package controlador;

import modelo.Empleado;
import modelo.IReportable;
import modelo.Movimiento;
import modelo.Persona;
import modelo.Venta;

import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

public class ReporteController {
    private VentaController ventaController;
    private GastoFijoController gastoController;
    private PersonaController personaController;

    public ReporteController(VentaController ventaController, GastoFijoController gastoController,
                             PersonaController personaController) {
        this.ventaController = ventaController;
        this.gastoController = gastoController;
        this.personaController = personaController;
    }

    public List<Movimiento> obtenerMovimientos() throws SQLException {
        List<Movimiento> movimientos = new ArrayList<>();
        movimientos.addAll(ventaController.listar());
        movimientos.addAll(gastoController.listar());
        return movimientos;
    }

    public double calcularIngresos(List<Movimiento> movimientos) {
        double total = 0;
        for (Movimiento m : movimientos) {
            if (m.esIngreso()) {
                total += m.calcularTotal();
            }
        }
        return total;
    }

    public double calcularEgresos(List<Movimiento> movimientos) {
        double total = 0;
        for (Movimiento m : movimientos) {
            if (!m.esIngreso()) {
                total += m.calcularTotal();
            }
        }
        return total;
    }

    public double calcularCobrado(List<Venta> ventas) {
        double total = 0;
        for (Venta v : ventas) {
            total += v.getMontoPagado();
        }
        return total;
    }

    public double calcularPendiente(List<Venta> ventas) {
        double total = 0;
        for (Venta v : ventas) {
            total += v.calcularDeuda();
        }
        return total;
    }

    public List<IReportable> obtenerPlanilla() throws SQLException {
        List<IReportable> planilla = new ArrayList<>();
        for (Persona p : personaController.listar()) {
            if (p instanceof Empleado) {
                planilla.add((Empleado) p);
            }
        }
        return planilla;
    }

    public double sumarReporte(List<IReportable> items) {
        double total = 0;
        for (IReportable r : items) {
            total += r.reporteMonto();
        }
        return total;
    }
}

package controlador;

import modelo.Cliente;
import modelo.Venta;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

public class VentaController {
    private Connection connection;

    private static final String SELECT_BASE =
            "SELECT v.Codigo_venta, v.Precio, v.Peso, v.Pago_pollo, v.Fecha, "
            + "p.codigo_persona, p.nombre, p.numero_telefono, p.zona_residencia, p.deuda_total "
            + "FROM ventas v INNER JOIN personas p ON v.Codigo_Cliente = p.codigo_persona ";

    public VentaController(Connection connection) {
        this.connection = connection;
    }

    private Venta construir(ResultSet rs) throws SQLException {
        Cliente cliente = new Cliente(rs.getString("codigo_persona"), rs.getString("nombre"),
                rs.getString("numero_telefono"), rs.getString("zona_residencia"), rs.getDouble("deuda_total"));
        return new Venta(rs.getInt("Codigo_venta"), rs.getString("Fecha"), cliente,
                rs.getDouble("Peso"), rs.getDouble("Precio"), rs.getDouble("Pago_pollo"));
    }

    public List<Venta> listar() throws SQLException {
        List<Venta> lista = new ArrayList<>();
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(SELECT_BASE + "ORDER BY v.Fecha, v.Codigo_venta");
            rs = ps.executeQuery();
            while (rs.next()) {
                lista.add(construir(rs));
            }
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
        return lista;
    }

    public Venta buscar(int codigoVenta) throws SQLException {
        ResultSet rs = null;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(SELECT_BASE + "WHERE v.Codigo_venta = ?");
            ps.setInt(1, codigoVenta);
            rs = ps.executeQuery();
            if (rs.next()) {
                return construir(rs);
            }
            return null;
        } finally {
            if (rs != null) {
                rs.close();
            }
            if (ps != null) {
                ps.close();
            }
        }
    }

    public int registrar(Venta venta) throws SQLException {
        String sql = "INSERT INTO ventas (Precio, Peso, Pago_pollo, Fecha, Codigo_Cliente) VALUES (?, ?, ?, ?, ?)";
        int filas;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setDouble(1, venta.getPrecioKg());
            ps.setDouble(2, venta.getPesoKg());
            ps.setDouble(3, venta.getMontoPagado());
            ps.setString(4, venta.getFecha());
            ps.setString(5, venta.getCliente().getCodigoPersona());
            filas = ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
        if (filas > 0 && venta.calcularDeuda() > 0) {
            ajustarDeuda(venta.getCliente().getCodigoPersona(), venta.calcularDeuda());
        }
        return filas;
    }

    public int actualizar(Venta venta) throws SQLException {
        Venta anterior = buscar(venta.getId());
        if (anterior == null) {
            return 0;
        }
        String sql = "UPDATE ventas SET Precio = ?, Peso = ?, Pago_pollo = ?, Fecha = ?, Codigo_Cliente = ? "
                + "WHERE Codigo_venta = ?";
        int filas;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setDouble(1, venta.getPrecioKg());
            ps.setDouble(2, venta.getPesoKg());
            ps.setDouble(3, venta.getMontoPagado());
            ps.setString(4, venta.getFecha());
            ps.setString(5, venta.getCliente().getCodigoPersona());
            ps.setInt(6, venta.getId());
            filas = ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
        if (filas > 0) {
            ajustarDeuda(anterior.getCliente().getCodigoPersona(), -anterior.calcularDeuda());
            ajustarDeuda(venta.getCliente().getCodigoPersona(), venta.calcularDeuda());
        }
        return filas;
    }

    public int eliminar(int codigoVenta) throws SQLException {
        Venta anterior = buscar(codigoVenta);
        if (anterior == null) {
            return 0;
        }
        String sql = "DELETE FROM ventas WHERE Codigo_venta = ?";
        int filas;
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setInt(1, codigoVenta);
            filas = ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
        if (filas > 0) {
            ajustarDeuda(anterior.getCliente().getCodigoPersona(), -anterior.calcularDeuda());
        }
        return filas;
    }

    private void ajustarDeuda(String codigoCliente, double diferencia) throws SQLException {
        if (diferencia == 0) {
            return;
        }
        String sql = "UPDATE personas SET deuda_total = GREATEST(deuda_total + ?, 0) WHERE codigo_persona = ?";
        PreparedStatement ps = null;
        try {
            ps = connection.prepareStatement(sql);
            ps.setDouble(1, diferencia);
            ps.setString(2, codigoCliente);
            ps.executeUpdate();
        } finally {
            if (ps != null) {
                ps.close();
            }
        }
    }
}

package modelo;

public class Cliente extends Persona {
    private String zonaResidencia;
    private double deudaTotal;

    public Cliente(String codigoPersona, String nombre, String numeroTelefono,
                   String zonaResidencia, double deudaTotal) {
        super(codigoPersona, nombre, numeroTelefono);
        this.zonaResidencia = zonaResidencia;
        this.deudaTotal = deudaTotal;
    }

    public String getZonaResidencia() {
        return zonaResidencia;
    }

    public void setZonaResidencia(String zonaResidencia) {
        this.zonaResidencia = zonaResidencia;
    }

    public double getDeudaTotal() {
        return deudaTotal;
    }

    public void setDeudaTotal(double deudaTotal) {
        this.deudaTotal = deudaTotal;
    }

    public boolean tieneDeuda() {
        return deudaTotal > 0;
    }

    @Override
    public boolean esEmpleado() {
        return false;
    }

    @Override
    public String obtenerResumen() {
        return "Cliente | Zona: " + zonaResidencia + " | Deuda: S/ " + String.format("%.2f", deudaTotal);
    }

package vista;

import java.util.Scanner;

public class Lector {
    private Scanner sc;

    public Lector(Scanner sc) {
        this.sc = sc;
    }

    public int leerEntero(String mensaje) {
        while (true) {
            System.out.print(mensaje);
            String texto = sc.nextLine().trim();
            try {
                return Integer.parseInt(texto);
            } catch (NumberFormatException e) {
                System.out.println("Debe ingresar un numero entero.");
            }
        }
    }

    public int leerEnteroNoNegativo(String mensaje) {
        while (true) {
            int valor = leerEntero(mensaje);
            if (valor >= 0) {
                return valor;
            }
            System.out.println("El valor no puede ser negativo.");
        }
    }

    public double leerDecimal(String mensaje) {
        while (true) {
            System.out.print(mensaje);
            String texto = sc.nextLine().trim().replace(',', '.');
            try {
                double valor = Double.parseDouble(texto);
                if (valor < 0) {
                    System.out.println("El valor no puede ser negativo.");
                } else {
                    return valor;
                }
            } catch (NumberFormatException e) {
                System.out.println("Debe ingresar un numero (ejemplo: 25.50).");
            }
        }
    }

    public String leerTexto(String mensaje) {
        while (true) {
            System.out.print(mensaje);
            String texto = sc.nextLine().trim();
            if (!texto.isEmpty()) {
                return texto;
            }
            System.out.println("El texto no puede estar vacio.");
        }
    }

    public String leerFecha(String mensaje) {
        while (true) {
            String texto = leerTexto(mensaje);
            if (esFechaValida(texto)) {
                return texto;
            }
            System.out.println("Fecha no valida. Use AAAA-MM-DD, con mes de 01 a 12 y un dia que exista en ese mes.");
        }
    }

    private boolean esFechaValida(String texto) {
        if (!texto.matches("[0-9]{4}-[0-9]{2}-[0-9]{2}")) {
            return false;
        }
        int anio = Integer.parseInt(texto.substring(0, 4));
        int mes = Integer.parseInt(texto.substring(5, 7));
        int dia = Integer.parseInt(texto.substring(8, 10));
        if (mes < 1 || mes > 12 || dia < 1) {
            return false;
        }
        int[] diasPorMes = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        int maximo = diasPorMes[mes - 1];
        boolean bisiesto = (anio % 4 == 0 && anio % 100 != 0) || anio % 400 == 0;
        if (mes == 2 && bisiesto) {
            maximo = 29;
        }
        return dia <= maximo;
    }

    public boolean confirmar(String mensaje) {
        String respuesta = leerTexto(mensaje + " (s/n): ");
        return respuesta.equalsIgnoreCase("s");
    }
}

package vista;

import controlador.ReporteController;
import controlador.VentaController;
import modelo.IReportable;
import modelo.Movimiento;
import modelo.Venta;

import java.sql.SQLException;
import java.util.List;

public class ReporteView extends VistaBase {
    private ReporteController reporteController;
    private VentaController ventaController;

    public ReporteView(Lector lector, ReporteController reporteController, VentaController ventaController) {
        super(lector);
        this.reporteController = reporteController;
        this.ventaController = ventaController;
    }

    @Override
    public void mostrarMenu() throws SQLException {
        int opcion;
        do {
            System.out.println("\n--- REPORTES ---");
            System.out.println("1. Movimientos y ganancia del negocio");
            System.out.println("2. Planilla de empleados");
            System.out.println("0. Regresar");
            opcion = getLector().leerEntero("Seleccione una opcion: ");
            switch (opcion) {
                case 1: reporteGanancia(); break;
                case 2: reportePlanilla(); break;
                case 0: break;
                default: System.out.println("Opcion no valida.");
            }
        } while (opcion != 0);
    }

    private void reporteGanancia() throws SQLException {
        List<Movimiento> movimientos = reporteController.obtenerMovimientos();
        if (movimientos.isEmpty()) {
            System.out.println("No hay movimientos registrados.");
            return;
        }
        System.out.println("\nMovimientos del negocio:");
        for (Movimiento m : movimientos) {
            System.out.println("   " + m.mostrarMovimiento());
        }
        List<Venta> ventas = ventaController.listar();
        double ingresos = reporteController.calcularIngresos(movimientos);
        double egresos = reporteController.calcularEgresos(movimientos);
        double cobrado = reporteController.calcularCobrado(ventas);
        double pendiente = reporteController.calcularPendiente(ventas);
        System.out.println();
        imprimirLinea("Total vendido:", ingresos);
        imprimirLinea("   - Cobrado:", cobrado);
        imprimirLinea("   - Pendiente:", pendiente);
        imprimirLinea("Total gastos fijos:", egresos);
        imprimirLinea("Ganancia real (caja):", cobrado - egresos);
        imprimirLinea("Ganancia proyectada:", ingresos - egresos);
    }

    private void reportePlanilla() throws SQLException {
        List<IReportable> planilla = reporteController.obtenerPlanilla();
        if (planilla.isEmpty()) {
            System.out.println("No hay empleados registrados.");
            return;
        }
        System.out.println("\nPlanilla del mes:");
        for (IReportable r : planilla) {
            System.out.println("   " + r.reporteTexto() + ": " + formatoMonto(r.reporteMonto()));
        }
        imprimirLinea("Total planilla:", reporteController.sumarReporte(planilla));
    }

    private void imprimirLinea(String etiqueta, double monto) {
        System.out.println(String.format("%-24s S/ %10.2f", etiqueta, monto));
    }
}

package vista;

import controlador.PersonaController;
import controlador.VentaController;
import modelo.Cliente;
import modelo.Venta;

import java.sql.SQLException;
import java.util.List;

public class VentaView extends VistaBase {
    private VentaController ventaController;
    private PersonaController personaController;

    public VentaView(Lector lector, VentaController ventaController, PersonaController personaController) {
        super(lector);
        this.ventaController = ventaController;
        this.personaController = personaController;
    }

    @Override
    public void mostrarMenu() throws SQLException {
        int opcion;
        do {
            System.out.println("\n--- VENTAS ---");
            System.out.println("1. Registrar venta");
            System.out.println("2. Listar ventas");
            System.out.println("3. Buscar venta por codigo");
            System.out.println("4. Actualizar venta");
            System.out.println("5. Eliminar venta");
            System.out.println("0. Regresar");
            opcion = getLector().leerEntero("Seleccione una opcion: ");
            switch (opcion) {
                case 1: registrar(); break;
                case 2: listar(); break;
                case 3: buscar(); break;
                case 4: actualizar(); break;
                case 5: eliminar(); break;
                case 0: break;
                default: System.out.println("Opcion no valida.");
            }
        } while (opcion != 0);
    }

    private Cliente pedirCliente() throws SQLException {
        String codigo = getLector().leerTexto("Codigo del cliente: ");
        Cliente cliente = personaController.buscarCliente(codigo);
        if (cliente == null) {
            System.out.println("No existe un cliente con el codigo " + codigo + ".");
        }
        return cliente;
    }

    private double pedirMontoPagado(double total) {
        while (true) {
            double pagado = getLector().leerDecimal("Monto pagado: ");
            if (pagado <= total) {
                return pagado;
            }
            System.out.println("El monto pagado no puede ser mayor al total (" + formatoMonto(total) + ").");
        }
    }

    private void registrar() throws SQLException {
        Lector lector = getLector();
        Cliente cliente = pedirCliente();
        if (cliente == null) {
            return;
        }
        String fecha = lector.leerFecha("Fecha (AAAA-MM-DD): ");
        double peso = lector.leerDecimal("Peso en kg: ");
        double precio = lector.leerDecimal("Precio por kg: ");
        double total = peso * precio;
        System.out.println("Total de la venta: " + formatoMonto(total));
        double pagado = pedirMontoPagado(total);
        Venta venta = new Venta(0, fecha, cliente, peso, precio, pagado);
        int filas = ventaController.registrar(venta);
        mostrarResultado(filas, "Venta registrada correctamente.");
        if (filas > 0 && venta.calcularDeuda() > 0) {
            System.out.println("El cliente queda debiendo " + formatoMonto(venta.calcularDeuda())
                    + " por esta venta (se sumo a su deuda total).");
        }
    }

    private void listar() throws SQLException {
        List<Venta> lista = ventaController.listar();
        if (lista.isEmpty()) {
            System.out.println("No hay ventas registradas.");
            return;
        }
        System.out.println("\nTotal: " + lista.size() + " venta(s)");
        for (Venta v : lista) {
            v.mostrarVenta();
        }
    }

    private void buscar() throws SQLException {
        int codigo = getLector().leerEntero("Codigo de la venta: ");
        Venta v = ventaController.buscar(codigo);
        if (v == null) {
            System.out.println("No se encontro la venta " + codigo + ".");
        } else {
            v.mostrarVenta();
        }
    }

    private void actualizar() throws SQLException {
        Lector lector = getLector();
        int codigo = lector.leerEntero("Codigo de la venta a actualizar: ");
        Venta actual = ventaController.buscar(codigo);
        if (actual == null) {
            System.out.println("No se encontro la venta " + codigo + ".");
            return;
        }
        System.out.println("Datos actuales:");
        actual.mostrarVenta();
        System.out.println("Ingrese los nuevos datos:");
        Cliente cliente = pedirCliente();
        if (cliente == null) {
            return;
        }
        actual.setCliente(cliente);
        actual.setFecha(lector.leerFecha("Fecha (AAAA-MM-DD): "));
        actual.setPesoKg(lector.leerDecimal("Peso en kg: "));
        actual.setPrecioKg(lector.leerDecimal("Precio por kg: "));
        System.out.println("Nuevo total: " + formatoMonto(actual.calcularTotal()));
        actual.setMontoPagado(pedirMontoPagado(actual.calcularTotal()));
        mostrarResultado(ventaController.actualizar(actual), "Venta actualizada correctamente.");
    }

    private void eliminar() throws SQLException {
        Lector lector = getLector();
        int codigo = lector.leerEntero("Codigo de la venta a eliminar: ");
        Venta v = ventaController.buscar(codigo);
        if (v == null) {
            System.out.println("No se encontro la venta " + codigo + ".");
            return;
        }
        v.mostrarVenta();
        if (lector.confirmar("Confirma eliminar esta venta?")) {
            mostrarResultado(ventaController.eliminar(codigo), "Venta eliminada correctamente.");
        } else {
            System.out.println("Operacion cancelada.");
        }
    }
}

package vista;

import java.sql.SQLException;

public abstract class VistaBase {
    private Lector lector;

    public VistaBase(Lector lector) {
        this.lector = lector;
    }

    public Lector getLector() {
        return lector;
    }

    public abstract void mostrarMenu() throws SQLException;

    public void mostrarResultado(int filas, String mensajeExito) {
        if (filas > 0) {
            System.out.println(mensajeExito);
        } else {
            System.out.println("No se realizo ningun cambio.");
        }
    }

    public String formatoMonto(double monto) {
        return "S/ " + String.format("%.2f", monto);
    }
}

    codigo_persona   VARCHAR(20)   NOT NULL,
    nombre           VARCHAR(100)  NOT NULL,
    numero_telefono  VARCHAR(20)   NOT NULL,
    es_empleado      TINYINT(1)    NOT NULL,
    zona_residencia  VARCHAR(100)  DEFAULT NULL,
    deuda_total      DECIMAL(10,2) DEFAULT NULL,
    cargo            VARCHAR(50)   DEFAULT NULL,
    pago_por_dia     DECIMAL(10,2) DEFAULT NULL,
    dias_trabajados  INT           DEFAULT NULL,
    bono             DECIMAL(10,2) DEFAULT NULL,
    PRIMARY KEY (codigo_persona)
) ENGINE=InnoDB;

) ENGINE=InnoDB;
