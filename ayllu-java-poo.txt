public class Cliente extends Persona {
    private String zonaResidencia;
    private double deudaTotal;

    public Cliente(String nombre, String numeroTelefono, String codigoPersona, String zonaResidencia, double deudaTotal) {
        super(nombre, numeroTelefono, codigoPersona);
        setZonaResidencia(zonaResidencia);
        setDeudaTotal(deudaTotal);
    }

    

    public String getZonaResidencia() {
        return zonaResidencia;
    }
    public void setZonaResidencia(String zonaResidencia) {
        if (zonaResidencia != null && !zonaResidencia.equals("")) {
            this.zonaResidencia = zonaResidencia;
        } else {
            System.out.println("Debe ingresar una zona residencia");
        }
    }
    public double getDeudaTotal() {
        return deudaTotal;
    }
    public void setDeudaTotal(double deudaTotal) {
        if (deudaTotal >= 0) {
            this.deudaTotal = deudaTotal;
        } else  {
            System.out.println("La deuda no debe ser negativo");
        }
    }

    @Override
    public void mostrarDatos(){
        System.out.println("Nombre: " + getNombre());
        System.out.println("Telefono: " + getNumeroTelefono());
        System.out.println("Codigo: " + getCodigoPersona());
        System.out.println("Zona Residencia: " + getZonaResidencia());
        System.out.println("Deuda Total: " + getDeudaTotal());
    }
    @Override
    public String obtenerResumen(){
        return "Cliente: " + getNombre() + " | Zona: " + zonaResidencia + " | Deuda: " + deudaTotal;
    }

    
    public void aumentarDeuda(double monto){
        deudaTotal += monto;
    }
    public void reducirDeuda(double monto){
        deudaTotal -= monto;
        if (deudaTotal < 0){
            deudaTotal = 0;
        }
    }
    public boolean tieneDeuda(){
        return deudaTotal > 0;
    }
}

public class Empleado extends Persona implements IReportable {
    
    private String cargo;
    private double pagoPorDia;
    private int diasTrabajados;
    private double bono;

    

    public Empleado(String nombre, String numeroTelefono, String codigoPersona, String cargo, double pagoPorDia, int diasTrabajados, double bono) {

        
        super(nombre, numeroTelefono, codigoPersona);

        
        setCargo(cargo);
        setPagoPorDia(pagoPorDia);
        setDiasTrabajados(diasTrabajados);
        setBono(bono);
    }

    

    
    @Override
    public String reporteTexto() {
        return "Empleado: " + getNombre()
                + ", Cargo: " + cargo
                + ", Pago mensual: " + calcularPago();

    }

    
    @Override
    public int reporteMontoMes() {
        return (int) calcularPago();
    }

    
    @Override
    public String obtenerResumen() {
        return "Empleado: " + getNombre() + ", Cargo: " + cargo;

    }

    
    public void registrarDias(int dias) {
        if (dias > 0) {
            diasTrabajados += dias;
        }

    public void asignarBono(double bono) {
        if (bono >= 0) {
            this.bono = bono;
        }
    }

        return diasTrabajados * pagoPorDia + bono;
    }

    
    public String getCargo() {
        return cargo;

    public void setCargo(String cargo) {
        if (cargo != null && !cargo.equals("")) {

import java.time.LocalDate;
import java.util.Arrays;
import java.util.List;

public class GastoFijo extends Movimiento implements IReportable {
    private String tipo;
    private double monto;

    
    private static final List<String> TIPOS_VALIDOS = Arrays.asList(
            "implementos",
            "combustible y peajes",
            "agua y gas",
            "otros"
    );

    public GastoFijo(int id, LocalDate fecha, String descripcion, String tipo, double monto) {
        super(id, fecha, descripcion);
        setTipo(tipo);
        setMonto(monto);
    }

    
    public String getTipo() {
        return tipo;
    }

    public void setTipo(String tipo) {
        if (tipo != null && TIPOS_VALIDOS.contains(tipo.toLowerCase().trim())) {
            this.tipo = tipo.toLowerCase().trim();
        } else {
            throw new IllegalArgumentException("Tipo de gasto no válido. Tipos permitidos: " + TIPOS_VALIDOS);
        }
    }

    public double getMonto() {
        return monto;
    }

    public void setMonto(double monto) {
        if (monto >= 0) {
            this.monto = monto;
        }
    }

    
    @Override
    public double calcularTotal() {
        return monto;
    }

    @Override
    public boolean esIngreso() {
        return false;
    }

    
    @Override
    public String reporteTexto() {
        return "Gasto Fijo (" + tipo + "): " + getDescripcion() + " - Total: $" + monto;
    }

    @Override
    public int reporteMontoMes() {
        return (int) Math.round(monto);
    }
}

import java.util.ArrayList;

public class GestorNegocio {
    private ArrayList<Cliente> clientes = new ArrayList<>();
    private ArrayList<Empleado> empleados = new ArrayList<>();

    public Cliente buscarCliente(String codigo) {
        for (Cliente cliente : clientes) {
            if (cliente.getCodigoPersona().equals(codigo)) {
                return cliente;
            }
        }
        return null;
    }

    public void registrarCliente(String nombre, String telefono, String codigo, String zonaResidencial, double deuda) {
        if (nombre.equals("") || telefono.equals("") || codigo.equals("") || zonaResidencial.equals("") || deuda < 0){
            System.out.println("Error: Datos incompletos o invalidos");
        } else if (buscarCliente(codigo) != null) {
            System.out.println("Error: Cliente ya existe");
        } else {
            clientes.add(new Cliente(nombre, telefono, codigo, zonaResidencial, deuda));
            System.out.println("Cliente registrado");
        }
    }

    public void mostrarClientes() {
        if (clientes.isEmpty()) {
            System.out.println("No hay clientes registrados...");
        }
        else {
            for (Cliente cliente : clientes) {
                cliente.mostrarDatos();
            }
        }
    }

    public void eliminarCliente(String codigo) {
        Cliente encontrado = buscarCliente(codigo);
        if (encontrado == null) {
            System.out.println("No se encontró el cliente");
        } else {
            clientes.remove(encontrado);
            System.out.println("Cliente eliminado");
        }
    }

    public Empleado buscarEmpleado(String codigo) {
        for (Empleado empleado : empleados) {
            if (empleado.getCodigoPersona().equals(codigo)) {
                return empleado;
            }
        }
        return null;
    }

    
    public void registrarEmpleado(String nombre, String telefono, String codigo,
                                   String cargo, double pagoPorDia,
                                   int diasTrabajados, double bono) {
        if (nombre.equals("") || telefono.equals("") || codigo.equals("") || cargo.equals("") || pagoPorDia <= 0 || diasTrabajados < 0 || bono < 0) {
            System.out.println("Error: Datos incompletos o inválidos");
        } else if (buscarEmpleado(codigo) != null) {
            System.out.println("Error: Empleado ya existe");
        } else {
            empleados.add(new Empleado(nombre, telefono, codigo, cargo, pagoPorDia, diasTrabajados, bono));
            System.out.println("Empleado registrado");
        }
    }

    
    public void mostrarEmpleados() {
        if (empleados.isEmpty()) {
            System.out.println("No hay empleados registrados...");
        } else {
            for (Empleado empleado : empleados) {
                System.out.println(empleado.reporteTexto());
            }
        }
    }
}

import java.time.LocalDate;

public abstract class Movimiento {
    private int id;
    private LocalDate fecha;
    private String descripcion;

    

    public Movimiento(int id, LocalDate fecha, String descripcion) {
        setId(id);
        setFecha(fecha);
        setDescripcion(descripcion);
    }

    

    public int getId() {return id;}
    public LocalDate getFecha() {return fecha;}
    public String getDescripcion() {return descripcion;}

    

    public void setId(int id) {
        if (id > 0) {
            this.id = id;
        }
    }

    public void setFecha(LocalDate fecha) {
        if (fecha != null) {
            this.fecha = fecha;
        }
    }

    public void setDescripcion(String descripcion) {
        if (descripcion != null && !descripcion.equals("")) {
            this.descripcion = descripcion;
        }
    }

    

    public String mostrarMovimiento() {
        return "ID: " + id
                + ", Fecha: " + fecha
                + ", Descripción: " + descripcion;
    }

    

    public abstract double calcularTotal();
    public abstract boolean esIngreso();
}

import java.time.LocalDate;

public class Venta extends Movimiento implements IReportable {
    private Cliente cliente;
    private double pesoKg;
    private double precioKg;
    private double montoPagado;

    public Venta(int id, LocalDate fecha, String descripcion, Cliente cliente, double pesoKg, double precioKg, double montoPagado) {
        super(id, fecha, descripcion);
        this.cliente = cliente;

        if (pesoKg > 0) {
            this.pesoKg = pesoKg;
        }
        if (precioKg > 0) {
            this.precioKg = precioKg;
        }

        double total = calcularTotal();
        if (montoPagado >= 0 && montoPagado <= total) {
            this.montoPagado = montoPagado;
        } else {
            this.montoPagado = 0;
        }

        
        double deudaInicial = total - this.montoPagado;
        if (this.cliente != null && deudaInicial > 0) {
            this.cliente.aumentarDeuda((float) deudaInicial);
        }
    }

    
    public Cliente getCliente() {
        return cliente;
    }

    public void setCliente(Cliente cliente) {
        this.cliente = cliente;
    }

    public double getPesoKg() {
        return pesoKg;
    }

    public void setPesoKg(double nuevoPeso) {
        if (nuevoPeso > 0) {
            actualizarDeudaCliente(() -> this.pesoKg = nuevoPeso);
        }
    }

    public double getPrecioKg() {
        return precioKg;
    }

    public void setPrecioKg(double nuevoPrecio) {
        if (nuevoPrecio > 0) {
            actualizarDeudaCliente(() -> this.precioKg = nuevoPrecio);
        }
    }

    public double getMontoPagado() {
        return montoPagado;
    }

    public void setMontoPagado(double montoPagado) {
        double totalActual = calcularTotal();
        if (montoPagado >= 0 && montoPagado <= totalActual) {
            double diferencia = montoPagado - this.montoPagado;
            this.montoPagado = montoPagado;
            if (cliente != null) {
                if (diferencia > 0) {
                    cliente.reducirDeuda((float) diferencia);
                } else if (diferencia < 0) {
                    cliente.aumentarDeuda((float) -diferencia);
                }
            }
        }
    }

    
    public double calcularDeuda() {
        return calcularTotal() - montoPagado;
    }

    private void actualizarDeudaCliente(Runnable cambioAccion) {
        double deudaAnterior = calcularDeuda();
        cambioAccion.run();
        double nuevaDeuda = calcularDeuda();
        double diferencia = nuevaDeuda - deudaAnterior;

        if (cliente != null) {
            if (diferencia > 0) {
                cliente.aumentarDeuda((float) diferencia);
            } else if (diferencia < 0) {
                cliente.reducirDeuda((float) -diferencia);
            }
        }
    }

    
    @Override
    public double calcularTotal() {
        return pesoKg * precioKg;
    }

    @Override
    public boolean esIngreso() {
        return true;
    }

    
    @Override
    public String reporteTexto() {
        return "Venta realizada a " + (cliente != null ? cliente.getNombre() : "Sin cliente")
                + " por un total de $" + calcularTotal();
    }

    @Override
    public int reporteMontoMes() {
        return (int) Math.round(calcularTotal());
    }
}
